var tlCur=0;
function tlPlay(i){
  tlCur=i;
  var v=tlVideos[i];
  var s=v.start?'&start='+v.start:'';
  document.getElementById('tlIframe').src='https://www.youtube.com/embed/'+v.id+'?rel=0&modestbranding=1&autoplay=1'+s;
  document.getElementById('tlTitle').textContent=v.title;
  document.getElementById('tlDesc').textContent=v.desc;
  document.querySelectorAll('.tl-pl-item').forEach(function(el,idx){
    el.classList.toggle('tl-active',idx===i);
  });
  updateNavButtons();
}
function updateNavButtons(){
  var items = Array.from(document.querySelectorAll('.tl-pl-item'));
  var visibleItems = items.filter(function(el){return el.style.display !== 'none';});
  var currentActive = document.querySelector('.tl-pl-item.tl-active');
  var currentIndex = visibleItems.indexOf(currentActive);
  
  document.getElementById('tlPrev').disabled = currentIndex <= 0;
  document.getElementById('tlNext').disabled = currentIndex === -1 || currentIndex === visibleItems.length - 1;
  document.getElementById('tlNext').textContent = (currentIndex === visibleItems.length - 1) ? '✓ Completed' : 'Next video →';
}
function tlNav(direction){
  var items = Array.from(document.querySelectorAll('.tl-pl-item'));
  var visibleItems = items.filter(function(el){return el.style.display !== 'none';});
  var currentActive = document.querySelector('.tl-pl-item.tl-active');
  var currentIndex = visibleItems.indexOf(currentActive);
  
  var nextIndex = currentIndex + direction;
  if(nextIndex >= 0 && nextIndex < visibleItems.length){
    var globalIndex = parseInt(visibleItems[nextIndex].getAttribute('data-index'));
    tlPlay(globalIndex);
  }
}
function tlFilter(cat, btn){
  document.querySelectorAll('.tl-filter-btn').forEach(function(b){b.classList.remove('tl-filter-active');});
  btn.classList.add('tl-filter-active');
  
  var firstVisibleIdx = -1;
  document.querySelectorAll('.tl-pl-item').forEach(function(el, idx){
    var isVisible = (cat === 'all' || el.classList.contains('tl-cat-' + cat));
    el.style.display = isVisible ? 'flex' : 'none';
    if(isVisible && firstVisibleIdx === -1){
      firstVisibleIdx = idx;
    }
  });
  
  var currentActive = document.querySelector('.tl-pl-item.tl-active');
  if(currentActive && currentActive.style.display === 'none' && firstVisibleIdx !== -1){
    tlPlay(firstVisibleIdx);
  } else {
    updateNavButtons();
  }
}
